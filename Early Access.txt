local function T(box, id, text, default, key)
    box:AddToggle(id, {
        Text = text,
        Default = default,
        Callback = function(v) cfg[key] = v and true or false end,
    })
end
local function S(box, id, text, default, min, max, round, key)
    box:AddSlider(id, {
        Text = text,
        Default = default,
        Min = min,
        Max = max,
        Rounding = round or 0,
        Callback = function(v) cfg[key] = v end,
    })
end

-- ========== HELPERS ==========
local function getHRP()
    local c = LP.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end
local function getHum()
    local c = LP.Character
    return c and c:FindFirstChildOfClass("Humanoid")
end
local function getHead(char)
    if not char then return nil end
    return char:FindFirstChild("HitboxHead") or char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
end
local function isEnemy(p)
    if not p or p == LP or not p.Character then return false end
    local h = p.Character:FindFirstChildOfClass("Humanoid")
    if not h or h.Health <= 0 then return false end
    local ally = p.Character:FindFirstChild("_is_ally")
    if ally and ally.Value then return false end
    if LP.Team and p.Team and LP.Team == p.Team then return false end
    return true
end

local function getClosest(fov, range)
    Cam = workspace.CurrentCamera
    if not Cam then return nil, nil end
    local best, bestD = nil, 1e9
    local origin = Cam.CFrame.Position
    local cx, cy = Cam.ViewportSize.X * 0.5, Cam.ViewportSize.Y * 0.5
    for _, p in ipairs(Players:GetPlayers()) do
        if isEnemy(p) then
            local part = getHead(p.Character)
            if part then
                local dist = (part.Position - origin).Magnitude
                if dist <= range then
                    local sp, onS = Cam:WorldToViewportPoint(part.Position)
                    if onS then
                        local d = (Vector2.new(sp.X, sp.Y) - Vector2.new(cx, cy)).Magnitude
                        if d <= fov and d < bestD then
                            bestD = d
                            best = { plr = p, part = part, char = p.Character }
                        end
                    end
                end
            end
        end
    end
    if not best then return nil, nil end
    local pos = best.part.Position
    if cfg.silentPred or cfg.ragePred then
        local ehrp = best.char and best.char:FindFirstChild("HumanoidRootPart")
        local vel = ehrp and ehrp.AssemblyLinearVelocity or Vector3.zero
        local dist = (Cam.CFrame.Position - pos).Magnitude
        local lead = math.max(cfg.lead or 400, 80)
        local t = math.clamp(dist / lead, 0.04, 0.55)
        local str = cfg.ragePredStr or 1.8
        -- base velocity prediction
        local pred = vel * t * str
        -- anti-cheat / desync style movement (UE / KH / Sleepy style strafes)
        if cfg.ragePredExtra then
            local hist = getgenv()._archVelHist
            if not hist then hist = {}; getgenv()._archVelHist = hist end
            local key = best.plr and best.plr.UserId or 0
            hist[key] = hist[key] or vel
            -- blend current with last sample to catch sudden strafe changes
            local blended = hist[key]:Lerp(vel, 0.65)
            hist[key] = vel
            pred = blended * t * str
            -- if they look nearly still but micro-jitter (common with silent), push slightly along look
            if vel.Magnitude < 2 and ehrp then
                pred = pred + ehrp.CFrame.LookVector * (t * 6 * str)
            end
            -- high-speed peek guess
            if vel.Magnitude > 40 then
                pred = pred + vel.Unit * math.min(vel.Magnitude * 0.08, 12)
            end
        end
        pos = pos + pred
    end
    return best, pos
end

local flyConn = nil
local function startFly()
    if flyConn then flyConn:Disconnect() end
    flyConn = RunService.Heartbeat:Connect(function(dt)
        if not cfg.fly then return end
        local tHrp = getHRP()
        if not tHrp then return end
        local h = getHum()
        if h then h:ChangeState(Enum.HumanoidStateType.Physics) end
        local c = workspace.CurrentCamera
        if not c then return end
        local md = Vector3.zero
        if h and h.MoveDirection.Magnitude > 0 then
            local fl = Vector3.new(c.CFrame.LookVector.X, 0, c.CFrame.LookVector.Z)
            if fl.Magnitude > 0.001 then
                md = Vector3.new(h.MoveDirection.X, c.CFrame.LookVector.Y * h.MoveDirection:Dot(fl.Unit), h.MoveDirection.Z)
            else
                md = Vector3.new(h.MoveDirection.X, c.CFrame.LookVector.Y, h.MoveDirection.Z)
            end
        end
        if UIS:IsKeyDown(Enum.KeyCode.Space) then md = md + Vector3.new(0, 1, 0) end
        if UIS:IsKeyDown(Enum.KeyCode.LeftShift) then md = md - Vector3.new(0, 1, 0) end
        tHrp.AssemblyLinearVelocity = Vector3.zero
        tHrp.AssemblyAngularVelocity = Vector3.zero
        if md.Magnitude > 0 then
            tHrp.CFrame = tHrp.CFrame + (md.Unit * (cfg.flySpeed * dt))
        end
    end)
    bind(flyConn)
end
startFly()

    local need = cfg.aimbot or cfg.silent or cfg.ragebot or cfg.bowCombat or cfg.trigger or cfg.projectileMimic or cfg.projectileV3 or cfg.projectileV4 or cfg.slingBypass or cfg.bowBypass or cfg.daggerBypass
    if need and Cam then
        local fov = 200
        if cfg.aimbot then fov = cfg.aimbotFov end
        if cfg.silent then fov = math.max(fov, cfg.silentFov) end
        if cfg.ragebot then
            fov = cfg.rageFov >= 350 and 9999 or cfg.rageFov
        end
        if cfg.bowCombat then fov = math.max(fov, cfg.bowFov) end
        if cfg.trigger then fov = math.min(fov, cfg.triggerFov) end
        local range = cfg.ragebot and cfg.rageRange or (cfg.bowCombat and cfg.bowRange or 800)
        local t, pos = getClosest(fov, range)

        if t and pos then
            if cfg.aimbot or cfg.silent or cfg.bowCombat or cfg.ragebot then
                local smooth = cfg.aimbot and cfg.aimbotSmooth or 0.05
                Cam.CFrame = Cam.CFrame:Lerp(CFrame.lookAt(Cam.CFrame.Position, pos), math.clamp(1 - smooth, 0.1, 1))
            end

            if cfg.trigger then
                local now = tick()
                if now - trigLast >= cfg.triggerDelay then
                    trigLast = now
                    pcall(function() mouse1click() end)
                end
            end

            -- look at target head
            if (cfg.ragebot or cfg.bowCombat) and tickN % 3 == 0 then
                Cam.CFrame = CFrame.lookAt(Cam.CFrame.Position, pos)
            end

            -- teleport behind target
            if cfg.ragebot and cfg.rageTeleport and hrp then
                local nowtp = tick()
                if not getgenv()._neLastTP then getgenv()._neLastTP = 0 end
                if nowtp - getgenv()._neLastTP >= (cfg.rageTPDelay or 0.7) then
                    getgenv()._neLastTP = nowtp
                    pcall(function()
                        hrp.CFrame = CFrame.new(pos + Vector3.new(0, 0, 2.8))
                    end)
                end
            end

            if cfg.ragebot and cfg.voidSpam then
                local voidY = cfg.voidSpamY or -500
                pos = Vector3.new(pos.X, voidY, pos.Z)
                getgenv()._archVoidPulse = true
                getgenv()._archVoidY = voidY
            end

            if cfg.ragebot and cfg.rageAuto and hum and hum.Health > 0 then
                local now = tick()
                if now - rageLast >= cfg.rageRate then
                    rageLast = now
                    local item = char and char:FindFirstChild("EquippedItem")
                    -- resolve UseItem remote (Rivals path)
                    if not useItem then
                        pcall(function()
                            local remotes = game:GetService("ReplicatedStorage"):FindFirstChild("Remotes")
                            local fighter = remotes and remotes:FindFirstChild("Fighter")
                            useItem = fighter and fighter:FindFirstChild("UseItem")
                        end)
                    end
                    if useItem and item then
                        pcall(function()
                            local rot = t.char and t.char:FindFirstChild("HumanoidRootPart")
                            useItem:FireServer({
                                id = HttpService:GenerateGUID(false),
                                objectId = HttpService:GenerateGUID(false),
                                item = item.Name,
                                position = pos,
                                attackNum = math.random(1, 99999),
                                rotation = rot and rot.CFrame.Rotation or t.part.CFrame.Rotation,
                                one = Vector3.one,
                                packed = { "\0", "\1", "\2", "\3" },
                            })
                        end)
                        -- extra force for bow
                        if tostring(item.Name):lower():find("bow") then
                            pcall(function()
                                if item.StartShooting then item:StartShooting(pos) end
                                if item.Shoot then item:Shoot(pos) end
                            end)
                        end
                    end
                    pcall(function()
                        local tool = char:FindFirstChildOfClass("Tool")
                        if tool and tool.Activate then tool:Activate() end
                    end)
                    if cfg.customShootSound and cfg.shootSoundId and cfg.shootSoundId ~= "" then
                        pcall(function()
                            local s = Instance.new("Sound")
                            s.SoundId = cfg.shootSoundId
                            s.Volume = 1
                            s.Parent = hrp or workspace
                            s:Play()
                            game:GetService("Debris"):AddItem(s, 3)
                        end)
                    end
                end
            end

            -- projectile mimic
            if cfg.projectileMimic and tickN % 8 == 0 then
                pcall(function()
                    local ncheck = 0
                    for _, o in ipairs((tickN % 15 == 0) and workspace:GetChildren() or {}) do
                        if ncheck > 25 then break end
                        if o:IsA("BasePart") then
                            local n = string.lower(o.Name)
                            if n:find("projectile") or n:find("arrow") or n == "coreprojectile" then
                                ncheck += 1
                                local dir = pos - o.Position
                                if dir.Magnitude > 1 and dir.Magnitude < 800 then
                                    local spd = math.max(o.AssemblyLinearVelocity.Magnitude, 180)
                                    o.AssemblyLinearVelocity = dir.Unit * spd
                                end
                            end
                        end
                    end
                end)
            end

            -- Projectile V3: stronger redirect + snap CFrame toward head every few frames
            if cfg.projectileV3 and tickN % 3 == 0 then
                pcall(function()
                    local targetPos = pos
                    local ncheck = 0
                    local function steer(o)
                        if not o:IsA("BasePart") then return end
                        local n = string.lower(o.Name)
                        if not (n:find("projectile") or n:find("arrow") or n:find("bullet") or n == "coreprojectile" or n:find("rocket")) then
                            return
                        end
                        ncheck += 1
                        if ncheck > 40 then return end
                        local dir = targetPos - o.Position
                        local mag = dir.Magnitude
                        if mag < 1 or mag > 2000 then return end
                        local spd = math.max(o.AssemblyLinearVelocity.Magnitude, 220)
                        o.AssemblyLinearVelocity = dir.Unit * spd * 1.35
                        o.CFrame = CFrame.lookAt(o.Position, targetPos)
                        -- hard snap when close
                        if mag < 25 then
                            o.CFrame = CFrame.new(targetPos)
                            o.AssemblyLinearVelocity = dir.Unit * spd * 2
                        end
                    end
                    for _, o in ipairs((tickN % 15 == 0) and workspace:GetChildren() or {}) do
                        steer(o)
                        if ncheck > 40 then break end
                    end
                    -- also scan common folders
                    for _, folderName in ipairs({"Projectiles", "Effects", "Debris", "Thrown"}) do
                        local f = workspace:FindFirstChild(folderName)
                        if f then
                            for _, o in ipairs(f:GetChildren()) do
                                steer(o)
                            end
                        end
                    end
                end)
            end

            -- Projectile V4: actively shoot projectiles at target
            if cfg.projectileV4 and t and pos then
                local now = tick()
                if not getgenv()._archPV4Last then getgenv()._archPV4Last = 0 end
                if now - getgenv()._archPV4Last >= 0.08 then
                    getgenv()._archPV4Last = now
                    pcall(function()
                        -- fire UseItem / tool like rage
                        local item = char and char:FindFirstChild("EquippedItem")
                        if not useItem then
                            local remotes = game:GetService("ReplicatedStorage"):FindFirstChild("Remotes")
                            local fighter = remotes and remotes:FindFirstChild("Fighter")
                            useItem = fighter and fighter:FindFirstChild("UseItem")
                        end
                        if useItem and item then
                            useItem:FireServer({
                                id = HttpService:GenerateGUID(false),
                                objectId = HttpService:GenerateGUID(false),
                                item = item.Name,
                                position = pos,
                                attackNum = math.random(1, 99999),
                                rotation = t.part.CFrame.Rotation,
                                one = Vector3.one,
                                packed = { "\0", "\1", "\2", "\3" },
                            })
                        end
                        local tool = char and char:FindFirstChildOfClass("Tool")
                        if tool then
                            if tool.Activate then tool:Activate() end
                            pcall(function()
                                if tool:FindFirstChild("Shoot") then tool.Shoot:FireServer(pos) end
                                if tool:FindFirstChild("Fire") then tool.Fire:FireServer(pos) end
                            end)
                        end
                        -- redirect any live projectiles hard
                        for _, o in ipairs((tickN % 15 == 0) and workspace:GetChildren() or {}) do
                            if o:IsA("BasePart") then
                                local n = string.lower(o.Name)
                                if n:find("projectile") or n:find("arrow") or n:find("bullet") or n == "coreprojectile" then
                                    local dir = pos - o.Position
                                    if dir.Magnitude > 1 and dir.Magnitude < 2500 then
                                        local spd = math.max(o.AssemblyLinearVelocity.Magnitude, 280)
                                        o.AssemblyLinearVelocity = dir.Unit * spd * 1.6
                                        o.CFrame = CFrame.lookAt(o.Position, pos)
                                    end
                                end
                            end
                        end
                    end)
                end
            end
        end
    end
end))

-- Fake Position + Void / Void Spam / Evasion (freeze-safe: one wait per tick, light work)
task.spawn(function()
    local lastVoidSpam = 0
    while true do
        RunService.Heartbeat:Wait()
        local ok, err = pcall(function()
            local char = LP.Character
            if not char then return end
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if not hrp then return end

            local fp = getgenv()._archFP
            local needDesync = (fp and fp.on and fp.offset)
                or cfg.voidEnabled
                or cfg.voidSpamEnabled
                or cfg.evasion
            if not needDesync then
                local panel = getgenv()._archXYZPanel
                if panel then panel.Visible = false end
                return
            end

            local old = hrp.CFrame
            local off = Vector3.zero

            if fp and fp.on and fp.offset then
                off = off + fp.offset
            end

            if cfg.voidEnabled then
                local bx = math.min(cfg.voidX or 1e6, 5e7)
                local by = math.min(cfg.voidY or 1e6, 5e7)
                local bz = math.min(cfg.voidZ or 1e6, 5e7)
                local ox, oy, oz = 0, 0, 0
                if cfg.voidAxisX then ox = (math.random() - 0.5) * 2 * bx end
                if cfg.voidAxisY then oy = (math.random() - 0.5) * 2 * by end
                if cfg.voidAxisZ then oz = (math.random() - 0.5) * 2 * bz end
                if ox == 0 and oy == 0 and oz == 0 then
                    oy = -math.min(by, 1e5)
                end
                off = off + Vector3.new(ox, oy, oz)
            end

            if cfg.evasion then
                local gap = math.min(cfg.evasionGap or 50, 200)
                off = off + Vector3.new((math.random() - 0.5) * gap, 0, (math.random() - 0.5) * gap)
            end

            local spamNow = false
            if cfg.voidSpamEnabled then
                local now = tick()
                local interval = math.max((cfg.voidSpamInterval or 50) / 1000, 0.05)
                if now - lastVoidSpam >= interval then
                    lastVoidSpam = now
                    spamNow = true
                    local far = math.min(math.max(cfg.voidSpamXp or 1e7, 1e6), 5e7)
                    off = off + Vector3.new(
                        math.random(-far, far),
                        math.random(-far, far),
                        math.random(-far, far)
                    )
                end
            end

            if off.Magnitude > 0 then
                hrp.CFrame = old + off
                RunService.RenderStepped:Wait()
                if hrp and hrp.Parent then
                    hrp.CFrame = old
                end
            end

            local panel = getgenv()._archXYZPanel
            if panel then
                local show = cfg.voidSpamEnabled or cfg.voidEnabled
                panel.Visible = show and true or false
                if show then
                    local p = hrp.Position
                    local xl = panel:FindFirstChild("X")
                    local yl = panel:FindFirstChild("Y")
                    local zl = panel:FindFirstChild("Z")
                    if xl then xl.Text = ("X: %.0f"):format(p.X) end
                    if yl then yl.Text = ("Y: %.0f"):format(p.Y) end
                    if zl then zl.Text = ("Z: %.0f"):format(p.Z) end
                end
            end
        end)
        if not ok then
            -- swallow errors so loop never dies / never spins
        end
    end
end)
